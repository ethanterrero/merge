# M-23 Role-aware Trips tab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the driver-only Trips tab with one role-aware TripsScreen (Requests / Upcoming / Past) on prototype mock data, hiding blocked people.

**Architecture:** Pure view logic and a tiny session store live in `src/lib/trips.ts` (tested under Node). Read-only fixtures and the one session instance live in `src/data/mockTrips.ts`. `DriverRequestsScreen.tsx` stops being a screen and exports the driving sections and their data hooks; the new `TripsScreen.tsx` composes them with the passenger sections. `TabBar.tsx` computes the badge from the same pure functions. Every section takes the data layer's `QueryState`, so M-35, M-36 and M-47 swap hook bodies only.

**Tech Stack:** Expo SDK 54, React Native 0.81, React 19 (`useSyncExternalStore`), TypeScript strict, `node:test` via `tsx`.

**Spec:** `docs/superpowers/specs/2026-10-10-trips-tab-design.md` (approved 2026-10-10, all ten owner questions as recommended, plus "Build amendments").

## Review Focus

1. **Blocking from another screen mid-session**: blocking Jordan on the driver's request screen, or Marcus on Match detail, must remove them from Trips and drop the badge at once. Pinned by `gateOnBlocks` tests (Task 1) and the walkthrough step in Task 4.
2. **Default segment flicker**: the block list is still loading on Trips' first mount, so the count is 0 for a frame; the default must latch only after the list settles and must not jump after the last request is answered. Pinned in Task 3 (latch effect) and checked in Task 4.
3. **Accept then come back**: an accepted invite must survive a tab switch (store, not component state) and must not count toward the badge. Pinned by session-store tests (Task 1) and Task 4's walkthrough.
4. **Driver confirm landing**: `DriverConfirmScreen` resets to `driverRequests` with `tab: 'upcoming'`; it must land on Trips → Upcoming with the rider under the right date only. Pinned by the `drivingDays` tests (Task 1) and Task 4.
5. **Role switch with existing data**: a passenger-only member must not see the driving fixtures, and the seats pill must hide. Pinned by `onSides`/`sidesFor` tests (Task 1) and Task 4's per-role walkthrough.

test('unknown ids are a no-op and keep the snapshot reference', () => {

  const before = store.getSnapshot();
  let calls = 0;
  store.subscribe(() => calls++);
  store.withdraw('nope');
  store.acceptInvite('nope');
  store.declineInvite('nope');
  assert.equal(store.getSnapshot(), before);
  assert.equal(calls, 0);

test('listeners fire on change, snapshots change reference, unsubscribe stops them', () => {
  const store = createTripsSession(seed());
  const before = store.getSnapshot();
  let calls = 0;
  const off = store.subscribe(() => calls++);
  store.declineInvite('x');
  assert.equal(calls, 1);
  assert.notEqual(store.getSnapshot(), before);
  off();
  store.declineInvite('y');
  assert.equal(calls, 1);
});

/**
 * Hide blocked people. Until the block list is ready the section reads `loading`, so
 * a blocked person is never shown even briefly; if it failed, the section shows the
 * generic error. `keep` returns the item (possibly trimmed) or null to drop it.
 */
export function gateOnBlocks<T>(
  state: QueryState<T[]>,
  blocked: BlockGate,
  keep: (item: T, isBlocked: BlockGate['isBlocked']) => T | null,
): QueryState<T[]> {
  if (state.status === 'error') return state;
  if (blocked.status === 'error') return { status: 'error', error: { kind: 'unknown', message: GENERIC_ERROR } };
  if (blocked.status !== 'ready') return { status: 'loading' };
  if (state.status !== 'success') return state;
  const kept = state.data.map((item) => keep(item, blocked.isBlocked)).filter((item): item is T => item !== null);
  return kept.length ? { ...state, data: kept } : { status: 'empty', refreshing: state.refreshing };
}

/**
 * My driving days, by date: the base days plus each accepted request under its own
 * date only (a day is created if I had none). Riders are never listed twice.
 */
export function drivingDays(
  base: readonly DrivingDay[],
  accepted: readonly { rideDate: string; rider: DrivingRider }[],
  leaveTime: string,
): DrivingDay[] {
  const byDate = new Map<string, DrivingDay>();
  for (const day of base) byDate.set(day.date, { ...day, riders: [...day.riders] });
  for (const { rideDate, rider } of accepted) {
    const day = byDate.get(rideDate) ?? { date: rideDate, leaveTime, riders: [] };
    if (!day.riders.some((r) => r.id === rider.id)) day.riders = [...day.riders, rider];
    byDate.set(rideDate, day);
  }
  return [...byDate.values()].sort((a, b) => a.date.localeCompare(b.date));
}

export type TripsSessionState = { sent: readonly SentItem[]; invites: readonly ReceivedInvite[] };

/**
 * The prototype's Withdraw / Accept / Decline state, shared by TripsScreen and the
 * tab bar through useSyncExternalStore. Snapshots are immutable and keep their
 * reference until something changes. Unknown ids are a no-op. M-35 replaces it with
 * the invitations data module.
 */
export function createTripsSession(seed: TripsSessionState) {
  let state: TripsSessionState = { sent: [...seed.sent], invites: [...seed.invites] };
  const listeners = new Set<() => void>();
  const set = (next: TripsSessionState) => {
    state = next;
    listeners.forEach((listener) => listener());
  };
  return {
    getSnapshot: (): TripsSessionState => state,
    subscribe: (listener: () => void): (() => void) => {
      listeners.add(listener);
      return () => {
        listeners.delete(listener);
      };
    },
    withdraw(id: string) {
      if (!state.sent.some((item) => item.id === id && canWithdraw(item))) return;
      set({ ...state, sent: state.sent.filter((item) => item.id !== id) });
    },
    acceptInvite(id: string) {
      const invite = state.invites.find((item) => item.id === id);
      if (!invite) return;
      const accepted: SentItem = { ...invite, type: 'invite', sender: 'them', status: 'accepted' };
      set({ sent: [...state.sent, accepted], invites: state.invites.filter((item) => item.id !== id) });
    },
    declineInvite(id: string) {
      if (!state.invites.some((item) => item.id === id)) return;
      set({ ...state, invites: state.invites.filter((item) => item.id !== id) });
    },
  };
}

                  <InlineConfirm
                    question={withdrawQuestion(item)}
                    keepLabel={item.type === 'request' ? 'Keep request' : 'Keep invite'}
                    confirmLabel="Withdraw"
                    onKeep={() => setConfirming(null)}

              </Card>
            );
          })
        : null}
    </View>
  );
}
